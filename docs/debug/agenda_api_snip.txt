import pandas as pd
import io


def _inicio_meses_atras(fecha_str, meses_atras=1):
    """
    Dada una fecha DD/MM/YYYY, retorna el 01 de N meses atrás.
    Si no se puede parsear, retorna la original.
    """
    try:
        dt = datetime.strptime(str(fecha_str).strip(), "%d/%m/%Y")
        year = dt.year
        month = dt.month
        for _ in range(max(0, int(meses_atras))):
            month -= 1
            if month == 0:
                month = 12
                year -= 1
        return f"01/{month:02d}/{year}"
    except Exception:
        return fecha_str

@api_bp.route("/hc", methods=["POST"])
@require_api_key
def descargar_hc():
    print("\n" + "!"*60)
    print(">>> [BACKEND] RECIBIDA SOLICITUD /api/hc")
    try:
        data = {}
        print(f">>> [BACKEND] Content-Type: {request.content_type}")
        
        # Verificar si es una petición con archivo (multipart/form-data)
        if request.content_type and "multipart/form-data" in request.content_type:
            print(">>> [BACKEND] Procesando multipart/form-data...")
            data = request.form.to_dict()
            archivo = request.files.get("archivo_excel")
            
            if archivo and archivo.filename != '':
                print(f">>> [BACKEND] Archivo Excel detectado: {archivo.filename}")
                try:
                    df = pd.read_excel(archivo)
                    print(f">>> [BACKEND] Excel cargado: {len(df)} filas.")
                    
                    df.columns = [str(c).lower().strip() for c in df.columns]
                    col_cc = next((c for c in df.columns if c in ["cc", "cedula", "documento", "id"]), None)
                    cols_serv = [c for c in df.columns if "servicio" in c or "area" in c]
                    if not cols_serv:
                        cols_serv = [c for c in df.columns if c in ["servicio", "area"]]
                    
                    col_est = next((c for c in df.columns if c in ["estrategia", "preferencia", "tipo"]), None)
                    
                    # Nuevas columnas de fecha
                    col_f_ini = next((c for c in df.columns if c in ["fecha inicio", "fecha_inicio", "desde", "f_ini", "fecha_desde"]), None)
                    col_f_fin = next((c for c in df.columns if c in ["fecha fin", "fecha_fin", "hasta", "f_fin", "fecha_hasta"]), None)
                    
                    if not col_cc:
                        print(">>> [BACKEND ERROR] No se encontró columna de Cédula")
                        return jsonify({"error": "No se encontró la columna de Cédula (CC) en el Excel"}), 400
                    
                    pacientes = []
                    for idx, row in df.iterrows():
                        if pd.isna(row[col_cc]): continue
                        cc = str(row[col_cc]).strip()
                        if cc.endswith(".0"): cc = cc[:-2]
                        if not cc or cc.lower() == "nan": continue
                            
                        est = str(row[col_est]).strip().upper() if col_est and pd.notna(row[col_est]) else None
                        
                        # Fechas individuales
                        f_ini = str(row[col_f_ini]).strip() if col_f_ini and pd.notna(row[col_f_ini]) else None
                        f_fin = str(row[col_f_fin]).strip() if col_f_fin and pd.notna(row[col_f_fin]) else None
                        
                        # Limpiar fechas nan/"nan"
                        if f_ini and f_ini.lower() == "nan": f_ini = None
                        if f_fin and f_fin.lower() == "nan": f_fin = None
                        
                        # Formatear fechas si vienen como datetime de pandas
                        if f_ini:
                            try:
                                # Si es timestamp, convertir
                                if hasattr(row[col_f_ini], 'strftime'):
                                    f_ini = row[col_f_ini].strftime("%d/%m/%Y")
                            except: pass
                            
                        if f_fin:
                            try:
                                if hasattr(row[col_f_fin], 'strftime'):
                                    f_fin = row[col_f_fin].strftime("%d/%m/%Y")
                            except: pass


                        todos_servicios = []
                        
                        if not cols_serv:
                            todos_servicios.append(None)
                        else:
                            for c_serv in cols_serv:
                                val = row[c_serv]
                                if pd.notna(val):
                                    val_str = str(val).strip()
                                    sub_servicios = [s.strip() for s in val_str.split(',')]
                                    todos_servicios.extend(sub_servicios)
                        
                        todos_servicios = list(set([s for s in todos_servicios if s]))
                        if not todos_servicios: todos_servicios = [None]

                        for s_item in todos_servicios:
                            pacientes.append({
                                "cedula": cc,
                                "servicio": s_item,
                                "estrategia": est,
                                "fecha_inicio": f_ini,
                                "fecha_fin": f_fin
                            })
                    
                    print(f">>> [BACKEND] Total pacientes a procesar: {len(pacientes)}")
                    data["pacientes_excel"] = pacientes
                except Exception as ex:
                    print(f">>> [BACKEND ERROR] Fallo procesando Excel: {ex}")
                    return jsonify({"error": f"Error en Excel: {str(ex)}"}), 400
            else:
                print(">>> [BACKEND] No hay archivo Excel. Usando datos manuales.")
        else:
            print(">>> [BACKEND] Petición JSON directa.")
            data = request.get_json(force=True)

        print(">>> [BACKEND] Validando modelo HCRequest...")
        req = HCRequest.from_dict(data)
        print(">>> [BACKEND] Modelo válido. Creando Job...")
        job_id = crear_job_hc(req)
        print(f">>> [BACKEND] Job creado exitosamente: {job_id}")
        return jsonify({"job_id": job_id}), 202
    except Exception as e:
        print(f">>> [BACKEND ERROR CRÍTICO]: {str(e)}")
        import traceback
        traceback.print_exc()
        return jsonify({"error": str(e)}), 400
